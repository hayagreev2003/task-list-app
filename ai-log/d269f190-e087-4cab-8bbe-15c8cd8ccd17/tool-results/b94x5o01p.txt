12 │ export function findInFileDuplicates<T extends Dedupable>(items: T[]) {
13 │   const firstRowByKey = new Map<string, number>();
14 │   const unique: T[] = [];
15 │   const duplicates: { item: T; firstRowNumber: number }[] = [];
16 │

18 │     const key = dedupeKey(item.task.title, item.task.due_date);
19 │     const firstRowNumber = firstRowByKey.get(key);

23 │     } else {
24 │       duplicates.push({ item, firstRowNumber });
25 │     }
26 │   }

 36 │   const parsed = parseCsv(text);
 37 │   if (!parsed.ok) return parsed;
 38 │ 
 39 │   if (parsed.rows.length > MAX_DATA_ROWS) {
 40 │     return {
 41 │       ok: false,
 42 │       error: `The file has ${parsed.rows.length} rows. The limit is ${MAX_DATA_ROWS}.`,
 43 │     };
 44 │   }
 45 │ 
 46 │   const rejected: RejectedRow[] = [];
 47 │   const valid: ImportCandidate[] = [];
 48 │   for (const row of parsed.rows) {
 49 │     const result = validateTaskInput(row.fields);
 50 │     if (result.ok) valid.push({ rowNumber: row.rowNumber, cells: row.cells, task: result.value });
 51 │     else rejected.push({ rowNumber: row.rowNumber, cells: row.cells, reasons: result.errors });
 52 │   }
 53 │ 
 54 │   const { unique, duplicates } = findInFileDuplicates(valid);
 55 │   for (const { item, firstRowNumber } of duplicates) {
 56 │     rejected.push({ rowNumber: item.rowNumber, cells: item.cells, reasons: [duplicateOfRow(firstRowNumber)] });
 57 │   }

 59 │   return {

 61 │     plan: {
 62 │       headers: parsed.headers,

 4 │ const BOM = "﻿";

 6 │ function isBlank(cells: string[]): boolean {
 7 │   return cells.every((cell) => cell.trim() === "");
 8 │ }
 9 │ 
10 │ /**
11 │  * Parses CSV text into rows keyed by known column.
12 │  * Row numbers count every record (blank ones included), so they match what a spreadsheet shows
13 │  * even when a quoted cell spans several lines.
14 │  */
15 │ export function parseCsv(input: string): ParseResult {
16 │   const withoutBom = input.startsWith(BOM) ? input.slice(BOM.length) : input;
17 │   // Normalise line endings so files mixing CRLF and LF parse consistently.
18 │   const text = withoutBom.replace(/\r\n?/g, "\n");
19 │   if (text.trim() === "") return { ok: false, error: "The file is empty." };
20 │ 
21 │   const parsed = Papa.parse<string[]>(text, { header: false, skipEmptyLines: false, newline: "\n" });
22 │ 
23 │   const quoteError = parsed.errors.find((e) => e.type === "Quotes");
24 │   if (quoteError) {
25 │     const row = quoteError.row === undefined ? "" : ` near row ${quoteError.row + 1}`;
26 │     return { ok: false, error: `The file has an unclosed quote${row}, so it can't be read reliably.` };
27 │   }
28 │ 
29 │   const records = parsed.data;
30 │   // A trailing newline yields one final empty record; it isn't a row.
31 │   const last = records.at(-1);
32 │   if (records.length > 1 && last && last.length === 1 && last[0] === "") records.pop();
33 │ 
34 │   // Skip any blank lines before the header.
35 │   const headerIndex = records.findIndex((cells) => !isBlank(cells));
36 │   if (headerIndex === -1) return { ok: false, error: "The file is empty." };
37 │ 
38 │   const headers = records[headerIndex].map((h) => h.trim());
39 │   const columnIndex = new Map<CsvColumn, number>();
40 │   headers.forEach((header, index) => {
41 │     const key = header.toLowerCase() as CsvColumn;
42 │     if ((KNOWN_COLUMNS as readonly string[]).includes(key) && !columnIndex.has(key)) {
43 │       columnIndex.set(key, index);
44 │     }
45 │   });
46 │ 
47 │   const missing = REQUIRED_COLUMNS.filter((column) => !columnIndex.has(column));
48 │   if (missing.length > 0) {
49 │     const label = missing.length === 1 ? "column" : "columns";
50 │     return {
51 │       ok: false,
52 │       error: `Missing required ${label}: ${missing.join(", ")}. Expected columns: ${KNOWN_COLUMNS.join(", ")}.`,
53 │     };
54 │   }
55 │ 
56 │   const rows: ParsedRow[] = [];
57 │   let blankSkipped = 0;
58 │   for (let i = headerIndex + 1; i < records.length; i++) {
59 │     const cells = records[i];
60 │     if (isBlank(cells)) {
61 │       blankSkipped++;
62 │       continue;
63 │     }
64 │     const cellAt = (column: CsvColumn) => {
65 │       const index = columnIndex.get(column);
66 │       return index === undefined ? "" : (cells[index] ?? "");
67 │     };
68 │     rows.push({
69 │       rowNumber: i + 1,
70 │       cells,
71 │       fields: {
72 │         title: cellAt("title"),
73 │         due_date: cellAt("due_date"),
74 │         priority: cellAt("priority"),
75 │         notes: cellAt("notes"),
76 │       },
77 │     });
78 │   }
79 │ 
80 │   return { ok: true, headers, rows, blankSkipped };
81 │ }

 50 │ /** True when `value` is YYYY-MM-DD and names a day that exists on the calendar. */
 51 │ export function isValidIsoDate(value: string): boolean {
 52 │   const match = DATE_PATTERN.exec(value);
 53 │   if (!match) return false;
 54 │   const [year, month, day] = match.slice(1).map(Number);
 55 │   // Year 0 doesn't exist in Postgres's (proleptic Gregorian, AD) date type.
 56 │   if (year < 1) return false;
 57 │   const date = new Date(0);
 58 │   // setUTCFullYear avoids Date.UTC's 0–99 → 1900–1999 mapping.
 59 │   date.setUTCFullYear(year, month - 1, day);
 60 │   return (
 61 │     date.getUTCFullYear() === year &&
 62 │     date.getUTCMonth() === month - 1 &&
 63 │     date.getUTCDate() === day
 64 │   );
 65 │ }

 1 │ import { createServerClient } from "@supabase/ssr";
 2 │ import { NextResponse, type NextRequest } from "next/server";
 3 │ import { supabaseEnv } from "@/lib/supabase/env";
 4 │ 
 5 │ const PUBLIC_PATHS = ["/login"];
 6 │ 
 7 │ /**
 8 │  * Refreshes the Supabase session cookie and makes an optimistic redirect.
 9 │  * This is not the authorisation check: requireUser() and RLS are.
10 │  */
11 │ export async function proxy(request: NextRequest) {
12 │   let response = NextResponse.next({ request });
13 │   const { url, publishableKey } = supabaseEnv();
14 │ 
15 │   const supabase = createServerClient(url, publishableKey, {
16 │     cookies: {
17 │       getAll() {
18 │         return request.cookies.getAll();
19 │       },
20 │       setAll(cookiesToSet, headers) {
21 │         for (const { name, value } of cookiesToSet) request.cookies.set(name, value);
22 │         response = NextResponse.next({ request });
23 │         for (const { name, value, options } of cookiesToSet) response.cookies.set(name, value, options);
24 │         for (const [key, value] of Object.entries(headers ?? {})) response.headers.set(key, value);
25 │       },
26 │     },
27 │   });
28 │ 
29 │   // getClaims validates the JWT and refreshes an expired session (writing new cookies above).
30 │   const { data } = await supabase.auth.getClaims();
31 │   const signedIn = Boolean(data?.claims);
32 │   const { pathname } = request.nextUrl;
33 │   const isPublic = PUBLIC_PATHS.some((p) => pathname === p || pathname.startsWith(`${p}/`));
34 │ 
35 │   if (!signedIn && !isPublic) return redirectKeepingCookies(request, response, "/login");
36 │   if (signedIn && isPublic) return redirectKeepingCookies(request, response, "/tasks");
37 │   return response;
38 │ }
39 │ 
40 │ function redirectKeepingCookies(request: NextRequest, from: NextResponse, pathname: string) {
41 │   const target = request.nextUrl.clone();
42 │   target.pathname = pathname;
43 │   target.search = "";
44 │   const redirect = NextResponse.redirect(target);
45 │   for (const cookie of from.cookies.getAll()) redirect.cookies.set(cookie);
46 │   return redirect;
47 │ }

18 │     const { error } = await supabase.auth.signInWithPassword({ email, password });

20 │   } else {
21 │     if (password.length < 8) return { error: "Use a password of at least 8 characters.", email };
22 │     const { data, error } = await supabase.auth.signUp({ email, password });
23 │     // Generic on purpose: don't reveal whether an email already has an account.
24 │     if (error) return { error: "Couldn't create an account with these details. If you already have one, sign in.", email };
25 │     if (!data.session) return { message: "Check your email to confirm your account, then sign in.", email };
26 │   }
27 │ 
28 │   redirect("/tasks");
29 │ }
